// 2. Indian Rupee Formatter Utility (TypeScript)
export const formatINR = (value: number): string => {
  return new Intl.NumberFormat('en-IN', {
    style: 'currency',
    currency: 'INR',
    maximumFractionDigits: 0
  }).format(value);
};
// formatINR(2845210) => "₹28,45,210"
// formatINR(486500000) => "₹48.65 Cr" via custom Crores pipe
